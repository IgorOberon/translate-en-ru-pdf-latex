\renewcommand{\includegraphics}[2][]{%
  \IfFileExists{#2}{%
    \sbox\FigMeasureBox{\SavedIncludegraphics[#1]{#2}}%
    \ifdim\wd\FigMeasureBox>\linewidth
      \SavedIncludegraphics[width=\linewidth,keepaspectratio]{#2}%
    \else\ifdim\ht\FigMeasureBox>0.82\textheight
      \SavedIncludegraphics[height=0.82\textheight,keepaspectratio]{#2}%
    \else\usebox\FigMeasureBox\fi\fi}%
  {\fbox{\parbox{0.7\linewidth}{\centering\small [рисунок не найден]\\ \ttfamily\tiny \detokenize{#2}}}}}
