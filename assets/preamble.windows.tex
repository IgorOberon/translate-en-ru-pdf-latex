% Russian XeLaTeX preamble for EN->RU translated documents with mathematics.
% Compile with XeLaTeX (or LuaLaTeX):  latexmk -xelatex main.tex
% Requires TeX Live packages: fontspec, polyglossia, csquotes, cm-unicode,
%   amsmath, amssymb, mathtools, graphicx, array, booktabs, longtable,
%   microtype, hyperref.

\documentclass[11pt,oneside]{book}

% ---- Fonts & Unicode (XeLaTeX / LuaLaTeX) -----------------------------------
\usepackage{fontspec}
% The main font MUST cover Cyrillic. "Times New Roman" matches Computer Modern and ships
% with the cm-unicode package. If it is not installed, swap for one of:
%   Libertinus Serif  ·  PT Serif  ·  Noto Serif  ·  DejaVu Serif
\setmainfont{Times New Roman}
\setsansfont{Arial}
\setmonofont{Consolas}

% ---- Language, hyphenation, typography (Russian primary, English secondary) --
\usepackage{polyglossia}
\setmainlanguage{russian}
\setotherlanguage{english}
\usepackage{csquotes}      % proper « » guillemets for Russian

% ---- Mathematics ------------------------------------------------------------
\usepackage{amsmath}
\usepackage{amssymb}
\usepackage{mathtools}
\usepackage{mathrsfs}   % \mathscr в формулах (нужен извлечённой математике)
% Math stays in Computer Modern math, which pairs cleanly with Times New Roman text.

% ---- Figures, tables, links -------------------------------------------------
\usepackage{graphicx}
\graphicspath{{images/}{./}}    % extracted figures are copied to build/images/
% figures behave: oversized ones shrink to the text width/height, a missing
% file becomes a framed note instead of killing the build
\newsavebox\FigMeasureBox
\let\SavedIncludegraphics\includegraphics
\renewcommand{\includegraphics}[2][]{%
  \IfFileExists{#2}{%
    \sbox\FigMeasureBox{\SavedIncludegraphics[#1]{#2}}%
    \ifdim\wd\FigMeasureBox>\linewidth
      \SavedIncludegraphics[width=\linewidth,keepaspectratio]{#2}%
    \else\ifdim\ht\FigMeasureBox>0.82\textheight
      \SavedIncludegraphics[height=0.82\textheight,keepaspectratio]{#2}%
    \else\usebox\FigMeasureBox\fi\fi}%
  {\fbox{\parbox{0.7\linewidth}{\centering\small [рисунок не найден]\\ \ttfamily\tiny \detokenize{#2}}}}}
% figure + caption glued together (emitted by assets/figcaption.lua):
% centered image, no page break before the caption, caption as wide as the
% image itself. Uses its own box register so \FigMeasureBox logic above
% keeps working inside \sbox.
\newsavebox\FigCapBox
\newcommand{\FigWithCaption}[2]{%
  \begin{center}
    \sbox\FigCapBox{\includegraphics{#1}}%
    \usebox\FigCapBox\par\nopagebreak\vspace{2pt}
    {\small\parbox{\wd\FigCapBox}{#2}}%
  \end{center}}
\usepackage{array}
\usepackage{booktabs}
\usepackage{longtable}          % pandoc emits long tables
\usepackage{microtype}          % XeLaTeX: character protrusion (safe subset)
% running heads: a long chapter title must never collide with the page
% number — keep the book's look (italic marks) but truncate to 86% of the
% text width. Chapter opening pages keep the plain style (number below).
\usepackage{fancyhdr}
\usepackage{truncate}
\pagestyle{fancy}
\fancyhf{}
\renewcommand{\headrulewidth}{0pt}
\fancyhead[L]{\itshape\truncate{0.86\textwidth}{\leftmark}}
\fancyhead[R]{\itshape\thepage}
\usepackage[hidelinks]{hyperref}

% ---- pandoc compatibility (we feed pandoc-generated body fragments) ----------
% pandoc's standalone template normally defines these; we assemble the body
% ourselves, so provide the common one here to avoid "undefined control sequence".
\providecommand{\tightlist}{\setlength{\itemsep}{0pt}\setlength{\parskip}{0pt}}
% pandoc 3.x wraps every image in \pandocbounded and table widths in \real —
% both live in pandoc's own template, which we do not use. Provide no-op fallbacks.
\providecommand{\pandocbounded}[1]{#1}
\providecommand{\real}[1]{#1}
% pandoc-3 longtable: таблица без подписи получает \LTcaptype=none, и longtable
% требует счётчик 'none' — предоставим его, если ещё не определён
\makeatletter\AtBeginDocument{\@ifundefined{c@none}{\newcounter{none}}{}}\makeatother

% Russian tends to have long words; loosen line breaking to avoid overfull lines.
\sloppy

% Плашки-определения, восстановленные rescue_plaques.py (текстовые рамки,
% которые извлечение классифицировало картинками)
\newcommand{\plaquebegin}{\par\medskip\noindent\hrule\smallskip\begin{quote}\itshape}
\newcommand{\plaqueend}{\end{quote}\smallskip\hrule\medskip\par}
